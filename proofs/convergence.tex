% Theorem

\begin{theorem}
Let $\mathbf{x} := (\mathbf{y}, \mathbf{z})$ and define
\[
\mathbf{F}(\mathbf{x}) :=
\begin{pmatrix}
\mathbf{G}(\mathbf{y}, \mathbf{z}) \\
\mathbf{H}(\mathbf{y}, \mathbf{z})
\end{pmatrix},
\qquad
\mathbf{F} : \mathbb{R}^{n+m} \to \mathbb{R}^{n+m}.
\]
Let $(\mathbf{y}_0,\mathbf{z}_0)$ be given.

Let $\mathbf{x}^* = (\mathbf{y}^*,\mathbf{z}^*)$ be a solution,
$\mathbf F(\mathbf x^*) = \mathbf 0$, and write the blocks of the Jacobian at the solution as
\[
\mathbf A := \partial_{\mathbf y}\mathbf G(\mathbf x^*),\quad
\mathbf B := \partial_{\mathbf z}\mathbf G(\mathbf x^*),\quad
\mathbf C := \partial_{\mathbf y}\mathbf H(\mathbf x^*),\quad
\mathbf D := \partial_{\mathbf z}\mathbf H(\mathbf x^*).
\]
For a pseudo-time $\delta>0$, with $c=\delta^{-1}$ (and $c=0$ for a pure Newton step), define the
\emph{outer iteration matrix}
\[
\mathbf T_c := (\mathbf A + c\mathbf I)^{-1}\bigl(c\,\mathbf I + \mathbf B\mathbf D^{-1}\mathbf C\bigr)
= \mathbf I - (\mathbf A + c\mathbf I)^{-1}\bigl(\mathbf A - \mathbf B\mathbf D^{-1}\mathbf C\bigr).
\]
For pure Newton ($c=0$) this is $\mathbf A^{-1}\mathbf B\mathbf D^{-1}\mathbf C$.

Assume there exists an open neighbourhood $\mathcal U \subset \mathbb{R}^{n+m}$
of $\mathbf{x}^*$ such that:

\begin{enumerate}
\item $\mathbf{G},\mathbf{H} \in C^1(\mathcal U)$;

\item the partial Jacobians
\[
\partial_{\mathbf y}\mathbf G(\mathbf y,\mathbf z),
\qquad
\partial_{\mathbf z}\mathbf H(\mathbf y,\mathbf z)
\]
are uniformly nonsingular on $\mathcal U$;

\item there exist constants
$\delta_0^{(z)}, \delta_0^{(y)} > 0$ and
$\delta_{\max}^{(z)}, \delta_{\max}^{(y)} < \infty$
such that the adaptive pseudo-time steps defined below satisfy
\[
0 < \delta_{k,l}^{(z)} \le \delta_{\max}^{(z)},
\qquad
\delta_{\min}^{(y)} \le \delta_k^{(y)} \le \delta_{\max}^{(y)},
\]
for some $\delta_{\min}^{(y)} > 0$. Put
$\Delta := [\,1/\delta_{\max}^{(y)},\, 1/\delta_{\min}^{(y)}\,]$, so that every outer
inverse pseudo-time $c_k := (\delta_k^{(y)})^{-1}$ lies in the compact set $\Delta$;

\item all inner and outer iterates generated by the algorithm remain in
$\mathcal U$;

\item (\emph{inner block}) $\mathbf D$ is nonsingular, and
$\mathbf A + c\mathbf I$ is nonsingular for every $c \in \Delta$;

\item (\emph{joint spectral radius condition}) the family
$\mathcal T := \{\mathbf T_c : c \in \Delta\}$ has joint spectral radius strictly below one,
\[
\hat\rho(\mathcal T)
:= \lim_{N\to\infty}\;\sup_{c_1,\dots,c_N\in\Delta}
\bigl\|\mathbf T_{c_N}\cdots\mathbf T_{c_1}\bigr\|^{1/N} < 1 .
\]
\end{enumerate}
Condition~(6) is the form taken by the requirement $\rho(\mathbf A^{-1}\mathbf B\mathbf D^{-1}\mathbf C)<1$
when pseudo-time steps are added and allowed to vary. If $\Delta$ is a single point $\{c\}$ it
reduces to $\rho(\mathbf T_c)<1$, and for $c = 0$ to $\rho(\mathbf A^{-1}\mathbf B\mathbf D^{-1}\mathbf C)<1$.
A sufficient condition is $\sup_{c\in\Delta}\|\mathbf T_c\| < 1$ in some norm.

For each $k \ge 0$, define the inner pseudo-transient continuation iteration
$\{\mathbf{z}_k^{\,l}\}_{l \ge 0}$ by
\[
\left(
(\delta_{k,l}^{(z)})^{-1} \mathbf I
+
\partial_{\mathbf z}\mathbf H(\mathbf y_k, \mathbf z_k^{\,l})
\right)
(\mathbf z_k^{\,l+1} - \mathbf z_k^{\,l})
=
-
\mathbf H(\mathbf y_k, \mathbf z_k^{\,l}),
\]
with adaptive time-step
\[
\delta_{k,l}^{(z)}
=
\min\!\left(
\delta_0^{(z)}
\frac{\|\mathbf H(\mathbf y_k, \mathbf z_k^{\,0})\|}
     {\|\mathbf H(\mathbf y_k, \mathbf z_k^{\,l})\|},
\,
\delta_{\max}^{(z)}
\right).
\]

Define
\[
\mathbf z_k^*
:=
\lim_{l\to\infty} \mathbf z_k^{\,l},
\]
whose existence is guaranteed by pseudo-transient continuation theory.

The outer iteration for $\mathbf y$ is defined by
\[
\left(
(\delta_k^{(y)})^{-1} \mathbf I
+
\partial_{\mathbf y}\mathbf G(\mathbf y_k, \mathbf z_k^*)
\right)
(\mathbf y_{k+1} - \mathbf y_k)
=
-
\mathbf G(\mathbf y_k, \mathbf z_k^*),
\]
with adaptive time-step
\[
\delta_k^{(y)}
=
\min\!\left(
\delta_0^{(y)}
\frac{\|\mathbf F(\mathbf x_0)\|}
     {\|\mathbf F(\mathbf y_k,\mathbf z_k^*)\|},
\,
\delta_{\max}^{(y)}
\right).
\]

Then the convergence is \emph{local and geometric}: there exist $\varepsilon>0$, $C\ge 0$ and
$\kappa\in[0,1)$, independent of the pseudo-time sequence $(c_k)$ in $\Delta$, such that whenever
$\|\mathbf y_0-\mathbf y^*\|<\varepsilon$,
\[
\|\mathbf y_k - \mathbf y^*\| \le C\kappa^k\|\mathbf y_0-\mathbf y^*\|
\quad\text{for all }k,
\]
and consequently $\{(\mathbf y_k,\mathbf z_k^*)\}$ converges to
$\mathbf x^* = (\mathbf y^*,\mathbf z^*)$ with $\mathbf F(\mathbf x^*) = \mathbf 0$.
\end{theorem}
\begin{proof}
Fix $k$ and consider the inner problem for $\mathbf z$ with $\mathbf y_k$
held fixed. Define
\[
\mathbf H_{\mathbf y_k}(\mathbf z) := \mathbf H(\mathbf y_k,\mathbf z).
\]

By Assumption~(1), $\mathbf H_{\mathbf y_k} \in C^1$, and by
Assumption~(2), $\partial_{\mathbf z}\mathbf H(\mathbf y_k,\mathbf z)$ is
nonsingular in $\mathcal U$.

The inner iteration is precisely the pseudo-transient continuation method
applied to the nonlinear equation
\[
\mathbf H_{\mathbf y_k}(\mathbf z) = \mathbf 0,
\]
with adaptive pseudo-time step $\delta_{k,l}^{(z)}$ that is uniformly bounded
above and bounded away from zero by Assumption~(3).
Therefore, by the global convergence theorem for pseudo-transient continuation
\cite{kelley1998convergence}, the sequence $\{\mathbf z_k^{\,l}\}_{l\ge0}$
converges to a limit
\[
\mathbf z_k^* := \lim_{l\to\infty} \mathbf z_k^{\,l},
\]
satisfying
\[
\mathbf H(\mathbf y_k,\mathbf z_k^*) = \mathbf 0.
\]

Thus, for each outer iteration index $k$, the inner pseudo-transient
continuation defines a reduced state $\mathbf z_k^*$.

The Implicit Function Theorem (applicable since $\mathbf D$ is nonsingular) gives a
$C^1$ map $\boldsymbol\zeta$ near $\mathbf y^*$ with $\boldsymbol\zeta(\mathbf y^*)=\mathbf z^*$,
$\mathbf H(\mathbf y,\boldsymbol\zeta(\mathbf y))=\mathbf 0$, and the solution of
$\mathbf H(\mathbf y,\mathbf z)=\mathbf 0$ is locally unique, so
$\mathbf z_k^*=\boldsymbol\zeta(\mathbf y_k)$. Differentiating
$\mathbf H(\mathbf y,\boldsymbol\zeta(\mathbf y))=\mathbf 0$ at $\mathbf y^*$ yields
$\boldsymbol\zeta'(\mathbf y^*) = -\mathbf D^{-1}\mathbf C$.

\emph{The outer iteration is a switched iteration, not a fixed pseudo-transient problem.}
Since $\mathbf z_k^*=\boldsymbol\zeta(\mathbf y_k)$ changes with $k$, the outer step is
\emph{not} pseudo-transient continuation applied to one fixed equation
$\mathbf G_{\mathbf z_k^*}(\mathbf y)=\mathbf 0$, and the global PTC theorem cannot simply be
re-invoked for $\mathbf y$. Instead, write the outer step as
\[
\mathbf y_{k+1}=\boldsymbol\Phi_{c_k}(\mathbf y_k),\qquad
\boldsymbol\Phi_c(\mathbf y) :=
\mathbf y-\bigl(\hat{\mathbf A}(\mathbf y)+c\mathbf I\bigr)^{-1}\mathbf G(\mathbf y,\boldsymbol\zeta(\mathbf y)),
\]
where $\hat{\mathbf A}(\mathbf y):=\partial_{\mathbf y}\mathbf G(\mathbf y,\boldsymbol\zeta(\mathbf y))$ is
continuous with $\hat{\mathbf A}(\mathbf y^*)=\mathbf A$, and $c_k=(\delta_k^{(y)})^{-1}\in\Delta$.

\emph{Step 1: linearisation.} By the chain rule,
$\frac{d}{d\mathbf y}\mathbf G(\mathbf y,\boldsymbol\zeta(\mathbf y))\big|_{\mathbf y^*}
=\mathbf A+\mathbf B\boldsymbol\zeta'(\mathbf y^*)=\mathbf A-\mathbf B\mathbf D^{-1}\mathbf C$
(the Schur complement), and since $\mathbf G(\mathbf y^*,\mathbf z^*)=\mathbf 0$,
\[
\boldsymbol\Phi_c(\mathbf y^*)=\mathbf y^*,\qquad
\boldsymbol\Phi_c'(\mathbf y^*)
=\mathbf I-(\mathbf A+c\mathbf I)^{-1}(\mathbf A-\mathbf B\mathbf D^{-1}\mathbf C)=\mathbf T_c .
\]
Because $\Delta$ is compact, $c\mapsto(\mathbf A+c\mathbf I)^{-1}$ is bounded and continuous on
$\Delta$ (Assumption~(5)), so the linearisation error
$\|\boldsymbol\Phi_c(\mathbf y)-\mathbf y^*-\mathbf T_c(\mathbf y-\mathbf y^*)\|
\le\eta\|\mathbf y-\mathbf y^*\|$ holds with $\eta\to0$ as $\|\mathbf y-\mathbf y^*\|\to0$,
\emph{uniformly in $c\in\Delta$}.

\emph{Step 2: joint spectral radius gives a uniform block contraction.}
By definition of the limit in Assumption~(6) there is $N\ge1$ and $q\in[0,1)$ with
\[
\bigl\|\mathbf T_{c_N}\cdots\mathbf T_{c_1}\bigr\|\le q
\qquad\text{for all }c_1,\dots,c_N\in\Delta .
\]
(Individual spectral radii $\rho(\mathbf T_c)<1$ would \emph{not} suffice here: the matrices
$\mathbf T_c$ need not commute, and a product of matrices with spectral radius below one can
grow.)

\emph{Step 3: switched local convergence.} The idea: over $N$ steps the nonlinear iteration
is close to the \emph{linear} product $\mathbf T_{N-1}\cdots\mathbf T_0$, which contracts by Step~2;
repeating this block gives geometric decay.

Fix any $(c_k)\subset\Delta$ and write $\mathbf w_k:=\mathbf y_k-\mathbf y^*$,
$\mathbf T_k:=\mathbf T_{c_k}$, $K:=\sup_{c\in\Delta}\|\mathbf T_c\|$.

\emph{(a) One block of $N$ steps.} Take any $\eta\in(0,1]$, and let $\varepsilon$ be the radius from
Step~1 for this $\eta$; put $L:=K+2$ and assume $\|\mathbf w_0\|<\varepsilon/L^{N}$. As long as
$\|\mathbf w_j\|<\varepsilon$, Step~1 applies to the step $\mathbf y_j\mapsto\mathbf y_{j+1}$.
Define the linearisation error of this step,
\[
\mathbf e_j:=\boldsymbol\Phi_{c_j}(\mathbf y_j)-\mathbf y^*-\mathbf T_j\mathbf w_j,
\]
so that, by Step~1,
\[
\mathbf w_{j+1}=\mathbf T_j\mathbf w_j+\mathbf e_j,\qquad\|\mathbf e_j\|\le\eta\|\mathbf w_j\| .
\]
Hence $\|\mathbf w_{j+1}\|\le(K+\eta)\|\mathbf w_j\|\le L\|\mathbf w_j\|$, so by induction
$\|\mathbf w_j\|\le L^{j}\|\mathbf w_0\|\le L^{N}\|\mathbf w_0\|<\varepsilon$ for all $j\le N$; the
estimate is therefore valid throughout the block. Unrolling the recursion,
\[
\mathbf w_N=\mathbf T_{N-1}\cdots\mathbf T_0\,\mathbf w_0+\sum_{j=0}^{N-1}\mathbf T_{N-1}\cdots\mathbf T_{j+1}\,\mathbf e_j .
\]
The first term has norm $\le q\|\mathbf w_0\|$ by Step~2. Each of the $N$ error terms has norm
$\le K^{N-1-j}\,\eta L^{j}\|\mathbf w_0\|\le\eta L^{N}\|\mathbf w_0\|$. Therefore
\[
\|\mathbf w_N\|\le\bigl(q+N\eta L^{N}\bigr)\|\mathbf w_0\| .
\]
Now choose $\eta$ so small that $N\eta L^{N}\le(1-q)/2$. Then, with $q':=(1+q)/2<1$,
\[
\|\mathbf w_N\|\le q'\,\|\mathbf w_0\|
\qquad\text{whenever }\|\mathbf w_0\|<\varepsilon_0:=\varepsilon/L^{N}.
\]

\emph{(b) Many blocks.} After $N$ steps we are again within $\varepsilon_0$ of $\mathbf y^*$, and the
remaining sequence $(c_{N+k})_k$ is again in $\Delta$, so (a) applies again. Thus
$\|\mathbf w_{aN}\|\le(q')^{a}\|\mathbf w_0\|$. For $k=aN+j$ with $0\le j<N$, the growth bound in (a)
gives $\|\mathbf w_k\|\le L^{N}\|\mathbf w_{aN}\|$, and so, with $\kappa:=(q')^{1/N}<1$,
\[
\|\mathbf y_k-\mathbf y^*\|\le L^{N}(q')^{a}\|\mathbf w_0\|\le\frac{L^{N}}{q'}\,\kappa^{k}\,\|\mathbf y_0-\mathbf y^*\|
\qquad(\text{since }(q')^{a+1}=\kappa^{N(a+1)}\le\kappa^{k}).
\]
The constants $\varepsilon_0$, $C=L^N/q'$ and $\kappa$ do not depend on $(c_k)$, so the result covers
the adaptive rule for $\delta_k^{(y)}$, where $c_k$ depends on the iterates.

\emph{Step 4: the limit.}
Thus $\mathbf y_k\to\mathbf y^*$, and by continuity of $\boldsymbol\zeta$,
$\mathbf z_k^*=\boldsymbol\zeta(\mathbf y_k)\to\boldsymbol\zeta(\mathbf y^*)=\mathbf z^*$.
Since $\mathbf H(\mathbf y_k,\mathbf z_k^*)=\mathbf 0$ for all $k$ and
$\mathbf H$ is continuous by Assumption~(1), passing to the limit gives
$\mathbf H(\mathbf y^*,\mathbf z^*)=\mathbf 0$, and $\mathbf G(\mathbf y^*,\mathbf z^*)=\mathbf 0$
by the choice of $\mathbf x^*$. Therefore
\[
\mathbf F(\mathbf x^*) = \mathbf 0,
\]
which completes the proof.
\end{proof}

\begin{remark}[Condition~(6) cannot be dropped]
Take $n=m=1$, $\mathbf G=y+2z$, $\mathbf H=2y+z$, so $\mathbf x^*=\mathbf 0$ and
$(\mathbf A,\mathbf B,\mathbf C,\mathbf D)=(1,2,2,1)$. Assumptions (1)--(5) hold for every
bounded pseudo-time, yet $\mathbf T_c=(c+4)/(c+1)>1$ for every $c\ge0$, so
$y_{k+1}=\mathbf T_{c_k}y_k$ diverges (for $c=0$: $y_{k+1}=4y_k$). Without the spectral condition
the convergence claim is false; mere nonsingularity of $\partial_{\mathbf y}\mathbf G$ and
$\partial_{\mathbf z}\mathbf H$ is not enough.
\end{remark}

\begin{remark}[Scope]
The theorem is local, and the inner solves are taken exact ($\mathbf z_k^*$ is the limit of the
inner iteration); inexact inner solves, line search and step damping are not covered. A
machine-checked version of Steps~1--4 (for exact inner solves) is in \texttt{proofs/lean}
(\texttt{split\_newton\_variable\_pseudo\_time}; joint spectral radius in
\texttt{SplitNewton/JointSpectralRadius.lean}).
\end{remark}
